def arvo_aarre(pelaaja_id, yhteys):
    kursori = yhteys.cursor()
    global käyty_kentät

    # Tarkistetaan onko pelaajalla jo aarre
    kursori.execute("SELECT airport_ident, found FROM treasure WHERE player_id = %s", (pelaaja_id,))
    tulos = kursori.fetchone()
    if tulos is not None:
        return tulos[0]  # Palautetaan olemassa oleva aarre

    # Jos ei ole, arvotaan uusi aarre
    airport_rows = lentokenttää()
    airport_idents = [row[0] for row in airport_rows]

    # Hae jo käydyt kentät
    kursori.execute("SELECT käyty_kentät FROM game WHERE id = %s", (pelaaja_id,))
    tulos = kursori.fetchone()
    käyty_kentät = tulos[0].split(",") if tulos and tulos[0] else []
    mahdolliset = [k for k in airport_idents if k not in käyty_kentät]

    if not mahdolliset:
        mahdolliset = airport_idents  # jos kaikki käyty, valitaan satunnainen

    aarre_kenttä = random.choice(mahdolliset)

    # Tallennetaan treasure-tauluun
    kursori.execute("""
        INSERT INTO treasure (player_id, airport_ident, found)
        VALUES (%s, %s, %s)
        ON DUPLICATE KEY UPDATE airport_ident = %s, found = %s
    """, (pelaaja_id, aarre_kenttä, False, aarre_kenttä, False))

    yhteys.commit()

def pelaa_minipeli(player_id, yhteys, kysymys, oikea_vastaus, max_yritykset=3):
    print(kysymys)
    yritys = 0

    while yritys < max_yritykset:
        vastaus = input(f"Yritys {yritys + 1}/{max_yritykset}: ")
        yritys += 1

        if vastaus.strip().lower() == oikea_vastaus.strip().lower():
            print("✅ Oikein! Saat 1 pisteen.")
            päivitä_motivaatio_taso(player_id, yhteys, +1)
            return True
        else:
            print("❌ Väärin!")

    print("😢 Yritykset loppuivat! Hävisit ja menetät 1 pisteen.")
    päivitä_motivaatio_taso(player_id, yhteys, -1)
    return False

from geopy.distance import geodesic

def hae_koordinaatit(ICAO, yhteys):
    """Hakee lentokentän leveys- ja pituusasteet tietokannasta."""
    kursori = yhteys.cursor()
    kursori.execute("SELECT latitude_deg, longitude_deg FROM airport WHERE ident = %s", (ICAO,))
    tulos = kursori.fetchone()
    if tulos:
        return tulos[0], tulos[1]
    else:
        return None, None

def laske_etaisyys(lat1, lon1, lat2, lon2):
    """Laskee etäisyyden GeoPy:n avulla kilometreinä."""
    paikka1 = (lat1, lon1)
    paikka2 = (lat2, lon2)
    return geodesic(paikka1, paikka2).kilometers


def lentää(käyttäjä, pelaaja_id, yhteys):
    valid_kenttä = kaikki_lentokentät()
    kursori = yhteys.cursor()

    # 🔎 Hae nykyinen sijainti ja kuljettu matka
    kursori.execute("SELECT location, matkustettu_matka FROM game WHERE id = %s", (pelaaja_id,))
    tulos = kursori.fetchone()

    if tulos is None:
        print("❌ Pelaajaa ei löytynyt tietokannasta.")
        return

    nykyinen_sijainti, matkustettu_matka = tulos

    while True:
        ICAO = input("Syötä haluamasi lentokentän ICAO-koodi: ").upper()

        if ICAO in valid_kenttä:
            # 📍 Hae koordinaatit ja laske etäisyys
            lat1, lon1 = hae_koordinaatit(nykyinen_sijainti, yhteys)
            lat2, lon2 = hae_koordinaatit(ICAO, yhteys)

            if None not in (lat1, lon1, lat2, lon2):
                etaisyys = laske_etaisyys(lat1, lon1, lat2, lon2)
                matkustettu_matka += etaisyys
                print(f"🧭 Lensit {etaisyys:.0f} km. Kokonaismatka: {matkustettu_matka:.0f} km.")
            else:
                print("⚠️ Koordinaattitietoja ei löytynyt, etäisyyttä ei voitu laskea.")
                etaisyys = 0

            # 🗺️ Päivitä sijainti ja kokonaismatka tietokantaan
            sql = '''UPDATE game 
                     SET location = %s, screen_name = %s, matkustettu_matka = %s 
                     WHERE id = %s'''

            yhteys.commit()
